// Initialize variables and get user data
const htmlContent = Inputs.htmlContent;
const do_final_review  = Inputs.do_final_review 
let currUser = Noodl.Object.get("userProperties");
const user_id = currUser.data.data[0].id;
const supabase = Noodl.Variables.supabase;
const titleId = Inputs.titleId;
const citations = Inputs.citations
const today = new Date().toISOString(); // ISO format: "2024-01-15T10:30:00.000Z"
//console.log("Creatomg Title ID");


async function deleteCitations() {
const { error } = await supabase
  .from('Titles_citations')
  .delete()
  .eq('user_id',user_id  )
  .eq('title_id',titleId  )
}

async function insertCitations() {
  if (!citations || !Array.isArray(citations) || citations.length === 0) return;

  const rows = citations.map(c => ({
    title_id: titleId,
    user_id: user_id,
    citation_id: c.id,
    author: c.author || null,
    formatted_text: c.formatted_text || null,
    url: c.url || null,
    publication_date: c.publication_date || null
  }));

  const { data, error } = await supabase
    .from('Titles_citations')
    .insert(rows)
    .select();

  if (error) {
    console.error('Insert citations error:', error);
    throw error;
  }
  return data;
}



async function UpdateTitleRecord() {
    console.log('Updating html in Titles table. user_id:', user_id, ", TitleID:", titleId);
    const { data, error } = await supabase
        .from('Titles')
        .update({ htmlArticle: htmlContent, hook: Inputs.hook, thesis: Inputs.thesis, metaDescription: Inputs.metaDescription, excerpt: Inputs.excerpt, Title :Inputs.Title, status: "Generated"  })
        .eq('user_id', user_id)
        .eq('id', titleId )
        .select();
    



    if (error) {
        console.error('Error updating Titles record:', error);
        throw error;
    }
    if (do_final_review) {
        Outputs.FinalReview()
    } else {
        Outputs.Save()
    }

}


// Execute the main function
 UpdateTitleRecord();
deleteCitations()
insertCitations()
