function processKeywordsForWordPress(primary, secondary, focus) {
    const processed = {
        primary: [],
        secondary: [],
        all: []
    };
    
    // Process primary keywords
    if (typeof primary === 'string') {
        processed.primary = primary.split(',').map(k => k.trim()).filter(k => k);
    } else if (Array.isArray(primary)) {
        processed.primary = primary.map(k => typeof k === 'object' ? k.keyword : k).filter(k => k);
    }
    
    // Process secondary keywords
    if (typeof secondary === 'string') {
        processed.secondary = secondary.split(',').map(k => k.trim()).filter(k => k);
    } else if (Array.isArray(secondary)) {
        processed.secondary = secondary.map(k => typeof k === 'object' ? k.keyword : k).filter(k => k);
    }
    
    // Combine all keywords
    processed.all = [focus, ...processed.primary, ...processed.secondary].filter(k => k);
    
    return processed;
}

function processTagsForWordPress(keywords) {
    let tags = [];
    
    try {
        if (typeof keywords === 'string') {
            // Handle both JSON string and comma-separated string
            if (keywords.startsWith('[')) {
                tags = JSON.parse(keywords);
            } else {
                tags = keywords.split(',').map(k => k.trim()).filter(k => k);
            }
        } else if (Array.isArray(keywords)) {
            tags = keywords;
        }
    } catch (error) {
        console.error("Error processing tags:", error);
        tags = [];
    }
    
    return tags;
}

function calculateContentScore() {
    let score = 0;
    
    // Title optimization (30 points)
    if (metaTitle.length >= 30 && metaTitle.length <= 60) score += 15;
    if (focusKeyword && metaTitle.toLowerCase().includes(focusKeyword.toLowerCase())) score += 15;
    
    // Meta description optimization (20 points)
    if (metaDescription.length >= 120 && metaDescription.length <= 160) score += 10;
    if (focusKeyword && metaDescription.toLowerCase().includes(focusKeyword.toLowerCase())) score += 10;
    
    // Content optimization (30 points)
    const wordCount = calculateWordCount(postBody);
    if (wordCount > 1000) score += 15;
    if (focusKeyword && postBody.toLowerCase().includes(focusKeyword.toLowerCase())) score += 15;
    
    // Technical SEO (20 points)
    if (postBody.includes('<h2')) score += 5;
    if (postBody.includes('<h3')) score += 5;
    if (postBody.includes('<img')) score += 5;
    if (postBody.includes('href=')) score += 5;
    
    return Math.min(score, 100);
}

function buildKeywordAnalysis() {
    const cleanContent = postBody.replace(/<[^>]*>/g, '');
    const wordCount = calculateWordCount(postBody);
    
    return {
        focus_keyword: {
            keyword: focusKeyword,
            density: calculateKeywordDensity(cleanContent, focusKeyword, wordCount),
            count: countKeywordOccurrences(cleanContent, focusKeyword),
            in_title: metaTitle.toLowerCase().includes(focusKeyword.toLowerCase()),
            in_meta_description: metaDescription.toLowerCase().includes(focusKeyword.toLowerCase()),
            in_content: cleanContent.toLowerCase().includes(focusKeyword.toLowerCase())
        },
        primary_keywords: processKeywordsForWordPress(primaryKeywords, '', '').primary.map(keyword => ({
            keyword: keyword,
            density: calculateKeywordDensity(cleanContent, keyword, wordCount),
            count: countKeywordOccurrences(cleanContent, keyword)
        }))
    };
}

function buildReadabilityAnalysis() {
    const sentences = postBody.split(/[.!?]+/).filter(s => s.trim().length > 0);
    const words = postBody.replace(/<[^>]*>/g, '').split(/\s+/).filter(w => w.length > 0);
    const avgWordsPerSentence = words.length / sentences.length;
    
    return {
        sentences_count: sentences.length,
        words_count: words.length,
        avg_words_per_sentence: Math.round(avgWordsPerSentence * 100) / 100,
        readability_score: readabilityScore || calculateFleschScore(words.length, sentences.length),
        has_subheadings: postBody.includes('<h2') || postBody.includes('<h3'),
        has_lists: postBody.includes('<ul>') || postBody.includes('<ol>'),
        paragraph_count: (postBody.match(/<p>/g) || []).length
    };
}

function calculateWordCount(content) {
    return content.replace(/<[^>]*>/g, '').split(/\s+/).filter(w => w.length > 0).length;
}

function calculateReadingTime(content) {
    const wordCount = calculateWordCount(content);
    return Math.ceil(wordCount / 200); // Average reading speed
}

function calculateKeywordDensity(content, keyword, wordCount) {
    if (!keyword || !wordCount) return 0;
    const keywordCount = countKeywordOccurrences(content, keyword);
    return ((keywordCount / wordCount) * 100).toFixed(2);
}

function countKeywordOccurrences(content, keyword) {
    if (!keyword) return 0;
    const regex = new RegExp(`\\b${keyword.toLowerCase()}\\b`, 'gi');
    return (content.toLowerCase().match(regex) || []).length;
}

function calculateFleschScore(wordCount, sentenceCount) {
    // Simplified Flesch Reading Ease calculation
    const avgSentenceLength = wordCount / sentenceCount;
    const avgSyllablesPerWord = 1.5; // Approximation
    return Math.max(0, Math.min(100, 206.835 - (1.015 * avgSentenceLength) - (84.6 * avgSyllablesPerWord)));
}

async function publishEnhancedPost() {
    try {
        const enhancedPostData = buildEnhancedPostData();
        
        console.log("📤 Publishing post with enhanced SEO data");
        console.log("📊 Meta fields count:", Object.keys(enhancedPostData.meta).length);
        
        const response = await fetch(url, {
            method: 'POST',
            headers: headers,
            body: JSON.stringify(enhancedPostData)
        });
        
        if (!response.ok) {
            const errorText = await response.text();
            throw new Error(`HTTP error! status: ${response.status}, details: ${errorText}`);
        }
        
        const result = await response.json();
        
        console.log("✅ Post published successfully");
        console.log("🔗 Post ID:", result.id);
        console.log("🌐 Post URL:", result.link);
        
        Outputs.Response = result;
        Outputs.PostId = result.id;
        Outputs.PostUrl = result.link;
        
        // Verify SEO metadata was saved
        await verifySEOMetadata(result.id);
        
        // Get comprehensive post information
        const postInfo = await getEnhancedPostInfo(result.id);
        Outputs.PostInfo = postInfo;
        
        Outputs.status = `✅ Successfully published to ${domain} with full SEO optimization`;
        Outputs.Success();
        
        return result;
        
    } catch (error) {
        console.error("❌ Error publishing post:", error);
        Outputs.error = error.message;
        Outputs.status = `❌ Publication to ${domain} failed: ${error.message}`;
        throw error;
    }
}

Metric	What it measures	Typical source	How it’s used
total_search_volume	The raw number of searches (often per month) for the keywords associated with the idea. It’s an absolute count (e.g., 12 000 searches).	Directly from keyword‑research tools (Ahrefs, SEMrush, DataForSEO, etc.).	
traffic_potential_score	A derived score (usually 0‑100) that estimates how much organic traffic the idea could attract if it ranks well. It combines search volume, keyword difficulty, competition, and sometimes click‑through‑rate assumptions.	Calculated in the backend using the raw volume plus difficulty/competition metrics.	
Key distinction	total_search_volume is a quantity; traffic_potential_score is a quality‑adjusted rating that tells you “how valuable” that volume is after accounting for difficulty.
