
// Noodl function to call the additional knowledge enhancement:

const supabase = await Noodl.Variables.supabase;
const userData = await supabase.auth.getUser();
const userID = userData.data.user.id;
const rag_url = Inputs.rag_url;
const title_ids = Inputs.title_ids; // Array of title IDs or objects
const collection_name = Inputs.collection_name || 'additional_knowledge_collection';
const research_depth = Inputs.research_depth || 'standard'; // 'standard', 'deep', 'comprehensive'
const source_types = Inputs.source_types || ['all']; // Filter specific types
 

async function enhanceAdditionalKnowledge() {
    try {
        
        if (!rag_url) {
            throw new Error('rag_url input is required');
        }
        
        if (!title_ids || !Array.isArray(title_ids) || title_ids.length === 0) {
            throw new Error('title_ids input is required and must be a non-empty array');
        }
        
        console.log('🔍 Enhancing additional knowledge for titles...');
        console.log('📋 Processing titles:', title_ids);
        console.log('🎯 Research depth:', research_depth);
        console.log('👤 User ID:', userID);
        
        // Extract title ID values
        let titleIdValues = [];
        
        if (typeof title_ids[0] === 'string') {
            titleIdValues = title_ids;
        } else {
            titleIdValues = title_ids.map(titleObj => 
                titleObj.id || titleObj.title_id
            ).filter(id => id != null);
        }
        
        console.log('🎯 Extracted title IDs:', titleIdValues);
        
        // Use bulk endpoint if available
        const response = await fetch(`${rag_url}/enhance_additional_knowledge_bulk`, {
            method: 'POST',
            headers: {
                'Content-Type': 'application/json',
                'Accept': 'application/json'
            },
            body: JSON.stringify({
                title_ids: titleIdValues,
                user_id: userID,
                collection_name: collection_name,
                research_depth: research_depth,
                source_types: source_types,
                exclude_existing_urls: true
            })
        });

        if (!response.ok) {
            throw new Error(`HTTP error! status: ${response.status}`);
        }

        const result = await response.json();
        
        if (result.status === 'success') {
            const processing = result.processing_summary;
            const results = result.results;
            
            console.log(`✅ Additional knowledge enhancement completed:`);
            console.log(`   Eligible titles: ${processing.eligible_titles}`);
            console.log(`   Successful: ${processing.successful_enhancements}`);
            console.log(`   Additional documents: ${processing.total_additional_documents}`);
            
            // Main outputs
            Outputs.success();
            Outputs.user_id = userID;
            Outputs.processedAt = result.research_timestamp;
            Outputs.enhancementType = 'additional_knowledge';
            
            // Processing summary
            Outputs.totalTitlesRequested = processing.total_titles_requested;
            Outputs.eligibleTitles = processing.eligible_titles;
            Outputs.successfulEnhancements = processing.successful_enhancements;
            Outputs.failedEnhancements = processing.failed_enhancements;
            Outputs.successRate = processing.success_rate;
            Outputs.totalAdditionalDocuments = processing.total_additional_documents;
            
            // Detailed results
            Outputs.results = results;
            Outputs.successfulTitles = results.filter(r => r.status === 'success');
            Outputs.failedTitles = results.filter(r => r.status !== 'success');
            
            // Collection info
            Outputs.collectionName = collection_name;
            Outputs.researchDepth = research_depth;
            Outputs.sourceTypes = source_types;
            
            // Status messages
            if (processing.successful_enhancements === processing.eligible_titles && processing.total_additional_documents > 0) {
                Outputs.message = `✅ Enhanced ${processing.eligible_titles} titles with ${processing.total_additional_documents} additional documents`;
                Outputs.detailedMessage = `Successfully added additional knowledge to all eligible titles. Enhanced collection: '${collection_name}'.`;
                Outputs.overallStatus = 'complete_success';
            } else if (processing.successful_enhancements > 0) {
                Outputs.message = `⚠️ Partial success: Enhanced ${processing.successful_enhancements}/${processing.eligible_titles} titles`;
                Outputs.detailedMessage = `Added ${processing.total_additional_documents} additional documents to ${processing.successful_enhancements} titles.`;
                Outputs.overallStatus = 'partial_success';
            } else {
                Outputs.message = `❌ No additional knowledge could be added`;
                Outputs.detailedMessage = `No titles were successfully enhanced with additional knowledge.`;
                Outputs.overallStatus = 'failed';
            }
            
            // Next steps
            Outputs.nextSteps = result.next_steps;
            
            // Actionable flags
            Outputs.hasAdditionalDocuments = processing.total_additional_documents > 0;
            Outputs.canTestEnhancedRAG = processing.total_additional_documents > 0;
            Outputs.readyForUpdatedActions = processing.successful_enhancements > 0;
            
        } else {
            throw new Error(result.error || 'Additional knowledge enhancement failed');
        }
        
    } catch (error) {
        console.error('❌ Additional knowledge enhancement error:', error);
        
        Outputs.error = error.message;
        Outputs.user_id = userID;
        Outputs.overallStatus = 'error';
        Outputs.message = `Enhancement failed: ${error.message}`;
        Outputs.detailedMessage = `Failed to enhance additional knowledge: ${error.message}`;
        
        // Error defaults
        Outputs.totalTitlesRequested = title_ids ? title_ids.length : 0;
        Outputs.eligibleTitles = 0;
        Outputs.successfulEnhancements = 0;
        Outputs.totalAdditionalDocuments = 0;
        Outputs.results = [];
        Outputs.nextSteps = ['Check connection to RAG server', 'Verify title enhancement status'];
    }
}

// Run the function
enhanceAdditionalKnowledge();
