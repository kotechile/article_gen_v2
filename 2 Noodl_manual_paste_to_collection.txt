async function manageDocument() {
            console.log('docid: ', docid, ',content: ', content, 'url: ', url)
    try {
        if (docid) {
            // Update existing record
            console.log('docid ', docid, ' will be updated')
            const { data, error } = await supabase
                .from('lindex_documents')
                .update({
                    title: title,
                    parsedText: content,
                    url: url
                })
                .eq('user_id', userID)
                .eq('id', docid)
                .select();
                
            if (error) {
                console.error('Error updating document:', error);
                Outputs.Error = error;
                Outputs.Status = "Error updating document";
                Outputs.Failure();
                return;
            }
            
            console.log("Existing docid was updated");
            Outputs.Status = "Existing docid was updated";
                        // Query the updated record
            await querySupabase(supabase, "lindex_documents", docid, userID, Outputs);
            Outputs.docid = docid;
            Outputs.Success();
            
            return;
            
        } else {
            // Check if title and content are provided
            if (!title || !content) {
                console.log("Missing Title or Content. Record was not created");
                Outputs.Status = "Missing Title or Content. Record was not created";
                Outputs.Failure();
                return;
            }
            
            // Create new record
            const { data, error } = await supabase
                .from('lindex_documents')
                .insert({
                    title: title,
                    content: content,
                    url: url,
                    user_id: userID,
                    source_type: 'manual'
                })
                .select();
                
            if (error) {
                console.error('Error creating document:', error);
                Outputs.Error = error;
                Outputs.Status = "Error creating document";
                Outputs.Failure();
                return;
            }
            
            // Get the ID of the newly created record
            const newDocId = data[0].id;
            Outputs.Status = `New Document ${newDocId} was successfully created`;
            console.log("New Document", newDocId, "was successfully created");
            
            // Query the new record
            await querySupabase(supabase, "lindex_documents", newDocId, userID, Outputs);
            Outputs.docid = newDocId;
            Outputs.Success();
            return;
        }
    } catch (err) {
        console.error('Exception in document operation:', err);
        Outputs.Error = err;
        Outputs.Status = "Exception occurred during operation";
        Outputs.Failure();
        return;
    }
}

// Helper function to query Supabase for record details
async function querySupabase(supabase, tableName, docid, userID, Outputs) {
    try {
        // Query the record
        const { data: tableData, error } = await supabase
            .from(tableName)
            .select("*")
            .eq('id', docid)  // Changed from 'docid' to 'id'
            .eq('user_id', userID);
            
        if (error) {
            console.error(`Error fetching record from ${tableName}:`, error);
            Outputs.NotEmpty = false;
            Outputs.Items = [];
            Outputs.Failure();
            return;
        }
        
        Outputs.NotEmpty = tableData.length > 0;
        Outputs.Items = tableData;
        return;
    } catch (err) {
        console.error(`Exception querying ${tableName}:`, err);
        Outputs.NotEmpty = false;
        Outputs.Items = [];
        Outputs.Failure();
        return;
    }
}

    const content = Inputs.content;
    const title = Inputs.title;
    const url = Inputs.url;
    const userID = Inputs.userID;
